# ScheduleBoard WebSocket
# 下流の /api/scheduleboard/socket.io/ を、上流の /socket.io/ にマップする
# HTTP/2ではWebSocketが動作しないため、HTTP/1.1を強制
location ^~ /api/scheduleboard/socket.io/ {
    # HTTP/1.1を強制（WebSocketには必須）
    proxy_http_version 1.1;
    
    # WebSocketアップグレードヘッダー
    proxy_set_header Upgrade $http_upgrade;
    proxy_set_header Connection "upgrade";
    
    # 基本ヘッダー
    proxy_set_header Host $host;
    proxy_set_header X-Real-IP $remote_addr;
    proxy_set_header X-Forwarded-For $proxy_add_x_forwarded_for;
    proxy_set_header X-Forwarded-Proto $scheme;
    proxy_set_header X-Forwarded-Host $host;
    
    # キャッシュ無効化
    proxy_cache_bypass $http_upgrade;
    
    # ↓ これが肝：必ず末尾 "/" を付けて /socket.io/ を明示
    #    /api/scheduleboard/socket.io/(...) -> /socket.io/(...) で upstream(3000)へ
    proxy_pass http://127.0.0.1:3000/socket.io/;
    
    # WebSocket安定化（推奨）
    proxy_buffering off;
    proxy_read_timeout 86400;
    proxy_send_timeout 86400;
}
